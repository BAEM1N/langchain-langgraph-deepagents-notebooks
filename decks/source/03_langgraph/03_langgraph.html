01_introduction.ipynb노트북 기준
from typing import TypedDict
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    text: str
    word_count: int

def count_words(state: State) -> dict:
    return {"word_count": len(state["text"].split())}

builder = StateGraph(State)
builder.add_node("counter", count_words)
builder.add_edge(START, "counter")
builder.add_edge("counter", END)
graph = builder.compile()
print(graph.invoke({"text": "LangGraph 상태 그래프"}))

from typing import Annotated, TypedDict
import operator
from langgraph.types import Send

class FanOutState(TypedDict):
    items: list[str]
    results: Annotated[list[str], operator.add]

def dispatch(state: FanOutState) -> list[Send]:
    return [Send("worker", {"item": item}) for item in state["items"]]

def worker(state: dict) -> dict:
    return {"results": [f"처리: {state['item']}"]}

builder = StateGraph(FanOutState)
builder.add_node("worker", worker)
builder.add_conditional_edges(START, dispatch, ["worker"])
builder.add_edge("worker", END)

@task 결과는 future로 돌아오고 result()를 호출할 때 값이 필요해집니다

Functional API의 병렬성은 함수처럼 보이지만 즉시 값이 아니라 future를 다룬다는 점에서 출발합니다

@task

@entrypoint

전체 실행의 진입점과 체크포인터를 정의합니다.

03_functional_api.ipynb@task + @entrypoint
from langgraph.func import entrypoint, task
from langgraph.checkpoint.memory import InMemorySaver

@task
def fetch_data(url: str) -> str:
    return f"{url}에서 가져온 데이터"

@task
def process_data(data: str) -> str:
    return f"처리 완료: {data.upper()}"

@entrypoint(checkpointer=InMemorySaver())
def data_pipeline(url: str) -> str:
    raw = fetch_data(url).result()
    return process_data(raw).result()

# DeltaChannel 개념 예시 — 실제 실행은 langgraph>=1.2에서 확인하세요.
example = r'''
from typing import Annotated, Sequence
from typing_extensions import TypedDict
from langgraph.channels import DeltaChannel

def list_reducer(state: list, writes: Sequence[list]) -> list:
    # (current_state, sequence_of_writes) -> new_state
    result = list(state)
    for write in writes:
        result.extend(write)
    return result

class State(TypedDict):
    messages: Annotated[
        list[str],
        DeltaChannel(list_reducer, snapshot_frequency=5),
    ]
'''
